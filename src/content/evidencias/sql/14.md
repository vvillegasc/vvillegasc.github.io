title: "Diagnóstico por cargo: promedio del cargo, brecha %, antigüedad, historial y situación de cada empleado."
---

```sql
WITH base AS (
  SELECT e.employee_id,
         e.first_name || ' ' || e.last_name AS empleado,
         e.job_id,
         e.salary,
         TRUNC(MONTHS_BETWEEN(SYSDATE, e.hire_date) / 12) AS anios_servicio,
         ROUND(AVG(e.salary) OVER (PARTITION BY e.job_id), 2) AS prom_cargo,
         COUNT(*) OVER (PARTITION BY e.job_id) AS personas_cargo,
         CASE WHEN EXISTS (SELECT 1 FROM job_history h
                           WHERE h.employee_id = e.employee_id)
              THEN 'SI' ELSE 'NO' END AS tiene_historial
  FROM   employees e
)
SELECT empleado, job_id, salary, prom_cargo, personas_cargo,
       anios_servicio, tiene_historial,
       ROUND((prom_cargo - salary) / prom_cargo * 100, 2) AS brecha_pct,
       CASE
         WHEN salary < prom_cargo * 0.95 THEN 'POR DEBAJO'
         WHEN salary > prom_cargo * 1.05 THEN 'POR ENCIMA'
         ELSE 'EN PROMEDIO'
       END AS situacion
FROM   base
WHERE  personas_cargo > 1
ORDER BY job_id, brecha_pct DESC;
```
